class Solution {
    public List<List<Integer>> fourSum(int[] nums, int target) {
        List<List<Integer>> ans=new ArrayList<>();
        Arrays.sort(nums);
        HashSet<HashSet<Integer>> set=new HashSet<>();
        for(int i=0;i<nums.length;i++){
            int x=nums[i];
            for(int j=i+1;j<nums.length;j++){
                int y=nums[j];
                int l=j+1,r=nums.length-1;
                while(l<r){
                    long sum=(long)nums[l]+nums[r]+x+y;
                    if(sum==target){
                        List<Integer> tmp=new ArrayList<>();
                        HashSet<Integer> s=new HashSet<>();
                        s.add(x);s.add(y);s.add(nums[l]);s.add(nums[r]);
                        tmp.add(x);
                        tmp.add(y);
                        tmp.add(nums[l]);
                        tmp.add(nums[r]);
                        if(!set.contains(s)){
                            ans.add(tmp);
                            set.add(s);
                        }
                        l++;r--;
                        continue;
                    }else if(sum<target)l++;
                    else r--;
                }
            }
        }
        return ans;
    }
}

class Solution {
    public List<List<Integer>> fourSum(int[] nums, int target) {
        List<List<Integer>> ans=new ArrayList<>();
        Arrays.sort(nums);
        for(int i=0;i<nums.length;i++){
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            int x=nums[i];
            for(int j=i+1;j<nums.length;j++){
                if (j > i + 1 && nums[j] == nums[j - 1]) continue;
                int y=nums[j];
                int l=j+1,r=nums.length-1;
                while(l<r){
                    long sum=(long)nums[l]+nums[r]+x+y;
                    if(sum==target){
                        List<Integer> tmp=new ArrayList<>();
                        tmp.add(x);
                        tmp.add(y);
                        tmp.add(nums[l]);
                        tmp.add(nums[r]);
                        ans.add(tmp);
                        // skip duplicates
                        while (l < r && nums[l] == nums[l + 1]) l++;
                        while (l < r && nums[r] == nums[r - 1]) r--;
                        l++;r--;
                        continue;
                    }else if(sum<target)l++;
                    else r--;
                }
            }
        }
        return ans;
    }
}
